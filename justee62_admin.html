if (body.action === 'fix_evang_type') {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName('전도보고서');
  if (!sheet) return ok({message:'시트 없음'});

  var lastCol = sheet.getLastColumn();
  var headers = sheet.getRange(1, 1, 1, lastCol)
                     .getValues()[0];
  var docCol  = headers.indexOf('문서번호') + 1;
  var typeCol = headers.indexOf('전도유형') + 1;
  if (!docCol || !typeCol)
    return ok({message:'컬럼 없음'});

  // ── 이력 컬럼 확인 / 생성 ──────────────────
  var histHeader = '전도유형_수정이력';
  var histCol = headers.indexOf(histHeader) + 1;
  if (!histCol) {
    // 마지막 컬럼 다음에 새 헤더 추가
    histCol = lastCol + 1;
    sheet.getRange(1, histCol).setValue(histHeader);
  }

  // ── 행별 수정 처리 ─────────────────────────
  var data = sheet.getDataRange().getValues();
  var now  = Utilities.formatDate(
               new Date(), 'Asia/Seoul',
               'yyyy-MM-dd HH:mm:ss');
  var updated = 0;

  for (var i = 1; i < data.length; i++) {
    if (String(data[i][docCol-1]) !== String(body.docNumber))
      continue;

    var oldVal = data[i][typeCol-1];   // 현재 시트 값
    var prevHist = data[i][histCol-1]; // 기존 이력

    // 전도유형 수정
    sheet.getRange(i+1, typeCol).setValue(body.newType);

    // 이력 누적 (최신이 위)
    var entry = '[' + now + '] '
      + (body.oldType || oldVal)
      + ' → ' + body.newType;
    var newHist = prevHist
      ? entry + '
' + prevHist
      : entry;
    sheet.getRange(i+1, histCol).setValue(newHist);

    updated++;
  }
  return ok({updated: updated});
}
